  1. Общий статус кластера K3S                                                                                                                                                                                                                           
                                                                                                                                                                                                                                                         
  # Статус всех нод                                                                                                                                                                                                                                      
  kubectl get nodes -o wide                                                                                                                                                                                                                              
                                                                                                                                                                                                                                                         
  # Проверить метки на нодах                                                                                                                                                                                                                             
  kubectl get nodes --show-labels                                                                                                                                                                                                                        
                                                                                                                                                                                                                                                         
  # Проверить состояние всех компонентов кластера                                                                                                                                                                                                        
  kubectl get componentstatuses                                                                                                                                                                                                                          
                                                                                                                                                                                                                                                         
  2. Проверка приложения (namespace: ecommerce)                                                                                                                                                                                                          
                                                                                                                                                                                                                                                         
  # Все поды приложения                                                                                                                                                                                                                                  
  kubectl get pods -n ecommerce -o wide                                                                                                                                                                                                                  
                                                                                                                                                                                                                                                         
  # Статус deployments                                                                                                                                                                                                                                   
  kubectl get deployments -n ecommerce                                                                                                                                                                                                                   
                                                                                                                                                                                                                                                         
  # Статус сервисов                                                                                                                                                                                                                                      
  kubectl get svc -n ecommerce                                                                                                                                                                                                                           
                                                                                                                                                                                                                                                         
  # Логи конкретного пода (backend/frontend/nginx)                                                                                                                                                                                                       
  kubectl logs -n ecommerce deploy/backend                                                                                                                                                                                                               
  kubectl logs -n ecommerce deploy/frontend                                                                                                                                                                                                              
  kubectl logs -n ecommerce deploy/nginx                                                                                                                                                                                                                 
                                                                                                                                                                                                                                                         
  3. Проверка PostgreSQL/Patroni (namespace: database)                                                                                                                                                                                                   
                                                                                                                                                                                                                                                         
  # Все поды базы данных                                                                                                                                                                                                                                 
  kubectl get pods -n database -o wide                                                                                                                                                                                                                   
                                                                                                                                                                                                                                                         
  # Статус PostgreSQL кластера (Zalando Operator)                                                                                                                                                                                                        
  kubectl get postgresql -n database                                                                                                                                                                                                                     
                                                                                                                                                                                                                                                         
  # Детальная информация о кластере PostgreSQL                                                                                                                                                                                                           
  kubectl describe postgresql postgres-cluster -n database                                                                                                                                                                                               
                                                                                                                                                                                                                                                         
  # Посмотреть на каких нодах работают поды БД                                                                                                                                                                                                           
  kubectl get pods -n database -o wide                                                                                                                                                                                                                   
                                                                                                                                                                                                                                                         
  # Логи подов PostgreSQL                                                                                                                                                                                                                                
  kubectl logs -n database postgres-cluster-0                                                                                                                                                                                                            
  kubectl logs -n database postgres-cluster-1                                                                                                                                                                                                            
                                                                                                                                                                                                                                                         
  4. Проверка Patroni (кто primary, кто replica)                                                                                                                                                                                                         
                                                                                                                                                                                                                                                         
  # Зайти в под и проверить статус Patroni                                                                                                                                                                                                               
  kubectl exec -it -n database postgres-cluster-0 -- patronictl list                                                                                                                                                                                     
                                                                                                                                                                                                                                                         
  # Альтернатива через curl к Patroni API                                                                                                                                                                                                                
  kubectl exec -it -n database postgres-cluster-0 -- curl -s localhost:8008/cluster | jq                                                                                                                                                                 
                                                                                                                                                                                                                                                         
  5. Проверка endpoints БД                                                                                                                                                                                                                               
                                                                                                                                                                                                                                                         
  # Endpoints сервиса (куда идет трафик)                                                                                                                                                                                                                 
  kubectl get endpoints -n database                                                                                                                                                                                                                      
                                                                                                                                                                                                                                                         
  # Детали сервиса postgres-cluster                                                                                                                                                                                                                      
  kubectl describe svc postgres-cluster -n database                                                                                                                                                                                                      
                                                                                                                                                                                                                                                         
  ---                                                                                                                                                                                                                                                    
  Команды для тестирования отказоустойчивости                                                                                                                                                                                                            
                                                                                                                                                                                                                                                         
  Перед тестом - определить текущее состояние                                                                                                                                                                                                            
                                                                                                                                                                                                                                                         
  # 1. Определить где primary, где replica                                                                                                                                                                                                               
  kubectl exec -it -n database postgres-cluster-0 -- patronictl list                                                                                                                                                                                     
                                                                                                                                                                                                                                                         
  # 2. Посмотреть на каких ВМ (нодах) работают поды БД                                                                                                                                                                                                   
  kubectl get pods -n database -o wide                                                                                                                                                                                                                   
                                                                                                                                                                                                                                                         
  # 3. Проверить доступность сайта                                                                                                                                                                                                                       
  curl -I http://10.10.10.10:30080                                                                                                                                                                                                                       
                                                                                                                                                                                                                                                         
  Во время теста (когда выключили ВМ)                                                                                                                                                                                                                    
                                                                                                                                                                                                                                                         
  # Следить за состоянием подов БД в реальном времени                                                                                                                                                                                                    
  kubectl get pods -n database -o wide -w                                                                                                                                                                                                                
                                                                                                                                                                                                                                                         
  # Следить за событиями в namespace database                                                                                                                                                                                                            
  kubectl get events -n database -w                                                                                                                                                                                                                      
                                                                                                                                                                                                                                                         
  # Проверить статус ноды (покажет NotReady)                                                                                                                                                                                                             
  kubectl get nodes -o wide                                                                                                                                                                                                                              
                                                                                                                                                                                                                                                         
  # Проверить переключение primary/replica                                                                                                                                                                                                               
  kubectl exec -it -n database <оставшийся-под> -- patronictl list                                                                                                                                                                                       
                                                                                                                                                                                                                                                         
  После восстановления ВМ                                                                                                                                                                                                                                
                                                                                                                                                                                                                                                         
  # Проверить что нода вернулась                                                                                                                                                                                                                         
  kubectl get nodes -o wide                                                                                                                                                                                                                              
                                                                                                                                                                                                                                                         
  # Проверить восстановление подов                                                                                                                                                                                                                       
  kubectl get pods -n database -o wide                                                                                                                                                                                                                   
                                                                                                                                                                                                                                                         
  # Проверить синхронизацию replica                                                                                                                                                                                                                      
  kubectl exec -it -n database postgres-cluster-0 -- patronictl list
